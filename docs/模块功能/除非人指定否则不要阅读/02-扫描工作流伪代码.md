## 2. ScanFramePipeline（C2~C5）——单帧全链

```pseudo
CLASS ScanFramePipeline:
    PRIVATE ops:                                  // 构造一次，Execute 复用
        maskSep_      : LaserMarkingSeparationCUDA   // 预处理-01
        ccl_          : RegionAnalyzerCUDA           // ccl（标记点）
        imgSplit_/zernike_/imgMerge_ : ...           // 03/04/05
        undistortCpu_ : MarkerUndistortCPU           // 06（★查温度表1）
        ellipseFit_/markerMatch_/epipolarInt_/edgeMatch_/pointRecon_ : ...  // 07~11
        opticalFlow_  : MarkerOpticalFlowFuseCPU     // 配准-01（默认）
        frameFuse_    : FrameFuseCPU                 // 兜底
        steger_       : StegerExtractorCUDA(Flat)    // 激光04
        undistortCuda_: UndistortPointsCuda          // 05（★查温度表1）
        epipolarInterp_: EpipolarInterpCuda(false)   // 06（lineIdCheck=false）
        laserMatchScan_: LaserMatchScanCuda          // 07（★SetTempTable 表2）
        laserRecon_   : LaserReconstructCuda         // 08（Q）
        markerFuse_   : MarkerCloudFuseCPU           // 融合-02M
        laserFuseCuda_: LaserCloudFuseCuda           // 融合-02
        laserNormal_  : LaserCloudNormalCuda         // 融合-03
    PRIVATE:
        stream_    : cuda::Stream
        regState_  : RegistrationState               // §3
        tempTable1_ : StereoRectifyTempTable          // start 时注入
        obs_       : ScanObsCollector                // §5 旁路

    // ---------- 初始化（start 时一次）----------
    FUNCTION initialize(tempTable1, tempTable2, existingMarkers, obs):
        laserMatchScan_.SetTempTable(tempTable2)       // 一次性注入（T7 查表路径）
        IF existingMarkers 非空:
            markerFuse_.seed(existingMarkers)          // 首个落入者为代表（文档 §5.2）
            regState_.setGlobalMarkers(existingMarkers) // 配准靶标
        gpuWarmup(全算子)                              // 显存池 + kernel 预热

    // ---------- 单帧主入口 ----------
    FUNCTION processFrame(grayL, grayR: Mat, frameId, tempC) -> FrameResult:
        // ===== ① GPU 前端 =====
        sep = maskSep_.Execute(grayL, stream_)         // 产出 d_markingPointMask + d_laserMask
        sepR = maskSep_.Execute(grayR, stream_)        // 左右各一
        cclL = ccl_.Execute(sep.d_markingPointMask, stream_)
        cclR = ccl_.Execute(sepR.d_markingPointMask, stream_)

        // ===== ② 标记点 CPU 链（与标定 MarkerChain 同族，参数语义不同）=====
        markerOut = runMarkerChain(grayL, grayR, cclL.components, cclR.components, tempC)

        // ===== ③ 激光链（B 模式；A 模式整段跳过）=====
        laserOut = config_.enableLaser()
                   ? runLaserChain(sep.d_laserMask, sep.d_grayImage,
                                   sepR.d_laserMask, sepR.d_grayImage, tempC)
                   : 空

        // ===== ④ 配准 =====
        reg = runRegistration(markerOut)               // §3；产出 R/T + globalIds

        // ===== ⑤ 融合 =====
        fuseMarker(markerOut, reg)
        IF config_.enableLaser(): fuseLaser(laserOut, reg)

        // ===== ⑥ 组装输出 + obs 旁路 =====
        obs_.onFrame(reg.R, reg.T, markerOut.markerObs)  // GBA 输入（非 fusedPoints_）
        RETURN {frameId, reg.R, reg.T, markerOut.marker3D,
                laserOut.d_points3d(留 Device), tempC}

    // ---------- ② 标记点链（缺一不可的 9 算子，补齐现状缺口）----------
    FUNCTION runMarkerChain(grayL, grayR, compsL, compsR, tempC) -> MarkerChainOut:
        rect1 = tempTable1_.lookup(tempC)              ☠ CHECK nullptr→用最近有效温度/降级标记
        sL = imgSplit_.Execute(grayL, compsL.toRectList())
        sR = imgSplit_.Execute(grayR, compsR.toRectList())
        zL = [zernike_.Execute(img) FOR img IN sL.splitImages]; zR = 同右
        mL = imgMerge_.Execute(zL, compsL.rects); mR = 同右
        u  = undistortCpu_.Execute(mL.merged, mR.merged, mL.groupIds,
                                   rect1.R1/R2/P1/P2/Q)          // ★查表
        ellL/ellR = ellipseFit_(u)（按组批量）
        mm  = markerMatch_.Execute(ellL.centers, ellR.centers)
        eiL = epipolarInt_.Execute(ellL, mm.centerMatches); eiR = 同右
        em  = edgeMatch_.Execute(eiL.ellipseResults, eiR.ellipseResults, mm.centerMatches)
        pr  = pointRecon_.Execute(em, u.groupIds, mm.centerMatches)
        RETURN {pr.markerResults(3D+法线), ellL/R, mm}

    // ---------- ③ 激光链（Device 全程驻留，零下载）----------
    FUNCTION runLaserChain(dLaserMaskL, dGrayL, dLaserMaskR, dGrayR, tempC) -> LaserChainOut:
        rect1 = tempTable1_.lookup(tempC)
        out = {}
        FOR (mask, gray) IN [(dLaserMaskL,dGrayL), (dLaserMaskR,dGrayR)]:
            st  = steger_.Execute(gray, mask)                    // Flat：统一占位 line_id
            ud  = undistortCuda_.Execute(st.d_centerPoints, rect1.R1/P1)
            ep  = epipolarInterp_.Execute(ud.d_rectifiedPoints)  // lineIdCheck=false
            out[side] = ep
        lm  = laserMatchScan_.Execute(out.L.d_interpPoints, out.R.d_interpPoints,
                                      currentTemp=tempC)          // ★查表2；产出 line_id
        lr  = laserRecon_.Execute(lm.d_matchedL, lm.d_matchedR, lm.d_matched_line_ids, Q=rect1.Q)
        RETURN {d_points3d: lr.d_points3d}                        // 留 Device 直喂融合

    // ---------- ⑤ 融合 ----------
    FUNCTION fuseMarker(markerOut, reg):
        inputs = toMarkerFuseInputs(markerOut.markerResults, reg.globalIds)
        r = markerFuse_.Execute(inputs, reg.R, reg.T)             // 体素哈希跨帧累积
    FUNCTION fuseLaser(laserOut, reg):
        fr = laserFuseCuda_.Execute(laserOut.d_points3d, reg.R, reg.T)
        laserNormal_.Execute(fr.deviceContext, fr.newVoxelRange)  // 新建体素法线
```

## 3. RegistrationState + 配准（C4）

```pseudo
CLASS RegistrationState:
    prev_          : atomic<shared_ptr<PrevFrameState>>    // 单线程版退化为成员，接口不变
    globalMarkers_ : vector<MarkerCloudPoint>              // 靶标（existingMarkers 预填）
    isFirstFrame_  : bool

FUNCTION runRegistration(markerOut) -> RegOut:
    IF regState_.isFirstFrame:
        r = opticalFlow_.fuseFirstFrame(markerOut.markerResults,   // 首帧初始化
                                        regState_.globalMarkers)   // 挂 existing 时以已有点为靶标
        globalIds = [i FOR i IN 0..n]  或  bestIdx(匹配上已有点)     // 文档 §5.2
    ELSE:
        snap = regState_.snapshot()                     // load 最新（可跨几帧——调度版语义）
        r = opticalFlow_.Execute(markerOut.markerResults, snap)     // 光流主路径
        globalIds = r.getGlobalIds()
        IF NOT r.success OR r.matchedCount < 3:         // 兜底分支
            r2 = frameFuse_.Execute(toSet(markerOut), toSet(regState_.globalMarkers))
            globalIds = [-1 FOR ...]                    // 兜底帧 globalId=-1（保守，GBA 跳过先验）
            r = r2
    regState_.publish(toPrevFrameState(r, markerOut))   // store 原子覆盖
    RETURN {R: r.R, T: r.T, globalIds}
```

## 5. ScanWorkflow 主控（C1，两阶段，Q5）

```pseudo
CLASS ScanWorkflow IMPLEMENTS IWorkflow:
    PHASE: Streaming → GlobalOptim → Done | Error

    FUNCTION start(config: ScanConfig):
        ☠ CHECK 10 门禁（07 WorkflowManager 已查，此处防御性跳过）
        // 标定产物注入（D3 契约消费端）
        tempTable1 = calibStore.stereoRectifyTempTable()   ☠ CHECK hasData
        tempTable2 = calibStore.laserPlaneMapTempTable()   // B 模式才查
        // L2 累积器会话级 Clear（数据管理原则：扫描开始时由会话层显式触发——防上次会话残留）
        pipeline_.resetAccumulators()          // markerFuse_.Clear() + laserFuseCuda_.Clear() + obs_.clear()
        pipeline_.initialize(tempTable1, tempTable2, config.existingMarkers, obs_)
        IF config.enableLaser(): mcu.setLaserOn(true)       // B：激光 ON
        session_->startSession(config.projectName)
        // 注：ScanStarted 事件与状态转态由 07 WorkflowManager 统一发布（统一命令通道；
        //     本工作流不再自发 publishEvent——现状代码 ScanWorkflow.cpp:296 的自发事件在 H1 时移除）
        scanThread_ = spawn streamingLoop()

    FUNCTION streamingLoop():                // 单线程版（调度版由 07 Phase 3 替换）
        WHILE running_:
            IF state_ == Paused: interruptibleSleep(50ms, abort_); CONTINUE
            frame = frameBuffer_.popFrame(100ms)
            IF 无帧: CONTINUE
            tempC = hardwareMonitor_.getLatestTemperatures().相机均值
            TRY
                result = pipeline_.processFrame(frame.leftGray, frame.rightGray,
                                               frame.frameId, tempC)
                session_->onFrameProcessed()
                IF result.laser非空: session_->onFrameFused()
                notifyProgress("流式", detailJson={frameId, R/T 摘要})
            CATCH e: log; CONTINUE                          // 单帧失败不终止扫描

    FUNCTION stop():                          // ★Q5：跑完 GBA 才退出（阻塞语义）
        running_ = false; scanThread_.join()
        mcu.setLaserOn(false)
        phase_ = GlobalOptim
        IF config_.enableFinalBA:             // 恒 true
            gba_ = GlobalOptimStage(obs_.takeAll(), calibStore, regState_)
            gbaReport = gba_.run(progress=转发UI)        // 秒~分钟；display 冻结末帧
        session_->stopSession()
        publish(ScanStopped)
        phase_ = gbaReport.ok ? Done : Error
```

## 6. GlobalOptimStage（C7）

```pseudo
CLASS GlobalOptimStage:
    FUNCTION run(progress) -> GbaReport:
        // ===== GBA（09 算子 + 软先验改造）=====
        input = {frames: obsFrames,                 // 逐帧 {R_init,t_init,markerObs}
                 highPrecisionGlobalIds: 已有点 id 集,
                 X_existing: 已有点坐标, sigma: 0.001mm 量级}    // 文档 §5.3
        gbaOut = globalBACpu_.Execute(input)        ☠ CHECK 收敛/条件数
        // ===== 重融合（修正位姿回写）=====
        markerFuse_.Clear(); laserFuseCuda_.Clear()
        IF existingMarkers 非空: markerFuse_.seed(existing)    // 重灌已有点
        FOR f IN obsFrames:
            (R*, T*) = gbaOut.corrected(f.frameIndex)
            markerFuse_.Execute(f.markerObs, R*, T*)           // 修正位姿重跑
            IF B 模式: laserFuseCuda_.Execute(f.d_points3d 缓存, R*, T*)
        laserNormal_.Execute(全部新体素)
        // ===== 覆盖发布 =====
        pointCloudBuffer_.publishReplace(markerFuse_.fusedPoints,   // version++ 整体替换
                                         laserFuse_.deviceContext)
        RETURN {rms, 修正量统计, 软先验命中数}
```

> 注：重融合需要逐帧 d_points3d 缓存（GBA 前不清显存累积器则激光侧可只存逐帧输入引用；
> 内存/显存预算 = 25 分钟扫描 × 帧率 × 点数 —— **联调期需实测，超预算则降采样缓存**——开放点，见功能单元分解注记）。

## 7. 关键横切点备忘

| 点 | 规则 |
|---|---|
| 温度取值 | 每帧一次 `getLatestTemperatures()`（08 缓存，非阻塞）；lookup 失败 → 用最近有效温度 + Degraded 标记 |
| A 模式 GPU 前段 | mask_separation 无激光工况**未验证**（文档 §3.4）：编排层分支预留 `frontendA = maskExtract_ || maskSep_`，联调实测后定 |
| 兜底配准 globalId | =-1；GBA 的 highPrecisionGlobalIds 与软先验仅对 ≥0 的 id 生效 |
| obs 与融合分离 | ScanObsCollector 独立于 marker/laser 融合累积器——fusedPoints_ 已丢逐帧结构，GBA 只认 obs（文档 §4.2 铁律） |
| display 冻结 | GBA 期间渲染层不拉新快照即天然冻结；发布走 publishReplace 一次性 version++ |
| 单帧失败 | streamingLoop 内 try-catch 吞掉继续（扫描是长时任务，坏帧不终止）；连续 N 帧失败 → Degraded 上报 |
| 线程退场 | stop() 顺序：running_=false → join → 激光 OFF → GBA → session/publish——GBA 在 join 之后保证无新帧写入 obs_ |
