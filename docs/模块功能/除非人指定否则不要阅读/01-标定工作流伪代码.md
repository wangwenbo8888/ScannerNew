## 2. MarkerChain（B2）——标记点链编排壳 2-1~2-11

```pseudo
CLASS MarkerChain:
    // 成员算子：构造一次，Execute 复用（算子非线程安全，本类单线程使用）
    PRIVATE ops:
        maskExtract_   : MaskExtractCUDA        // 2-1
        frameFilter_   : FrameFilterCUDA        // 2-1b
        ccl_           : RegionAnalyzerCUDA     // 2-2
        imgSplit_      : ImageSplitCPU          // 2-3
        zernike_       : ZernikeEdgeCPU         // 2-4
        imgMerge_      : ImageMergeCPU          // 2-5
        undistortCpu_  : MarkerUndistortCPU     // 2-6
        ellipseFit_    : EllipseFitCPU          // 2-7
        markerMatch_   : MarkerMatchCPU         // 2-8
        epipolarInt_   : EpipolarIntersectCPU   // 2-9
        edgeMatch_     : EdgeMatchCPU           // 2-10
        pointRecon_    : PointReconstructCPU    // 2-11
    PRIVATE stream_   : cuda::Stream
    PRIVATE initialParams_ : InitialParams      // 从 CalibStore 注入，单一来源

    // ---------- GPU 前端（2-1 → 2-1b → 2-2）----------
    FUNCTION frontend(markerGrayL, markerGrayR: Mat) -> FrontendOutput
        // 灰度图双副本：Host Mat 已有，此处上传 Device（一次）
        r1 = maskExtract_.Execute(markerGrayL, stream_)    ☠ CHECK
        r2 = maskExtract_.Execute(markerGrayR, stream_)    ☠ CHECK
        // r 含 d_grayImage + d_cleanedMask（GpuMat 驻留）

        fL = frameFilter_.Execute(r1.d_cleanedMask, stream_)   // 2-1b
        fR = frameFilter_.Execute(r2.d_cleanedMask, stream_)
        IF NOT fL.isMarkerFrame OR NOT fR.isMarkerFrame:
            RETURN {frameType: LaserFrame}      // 激光线帧 → 调用方销毁，不走后续
                                                              // （标定阶段采集器应保证只投标记点帧，
                                                              //  此为防御）
        cL = ccl_.Execute(fL.d_cleanedMask, stream_)          // 2-2
        cR = ccl_.Execute(fR.d_cleanedMask, stream_)          ☠ CHECK
        RETURN {frameType: MarkerFrame,
                grayL/R: markerGrayL/R,                        // Host 副本给 CPU 链
                componentsL/R: cL.components}                  // 包围盒（ccl 内部已下载 Host）

    // ---------- CPU 链（2-3 ~ 2-11），左右各跑一遍后汇合 ----------
    FUNCTION compute(in: FrontendOutput) -> PointReconstructCPUResult
        // --- 左路 ---
        sL = imgSplit_.Execute(in.grayL, in.componentsL.toRectList())   // 2-3 深拷贝子图
        zL = [zernike_.Execute(img) FOR each img IN sL.splitImages]     // 2-4
        // --- 右路 ---
        sR = imgSplit_.Execute(in.grayR, in.componentsR.toRectList())
        zR = [zernike_.Execute(img) FOR each img IN sR.splitImages]
        // --- 汇合 ---
        mL = imgMerge_.Execute(zL, in.componentsL.rects)                // 2-5
        mR = imgMerge_.Execute(zR, in.componentsR.rects)
        u  = undistortCpu_.Execute(mL.mergedEdgePoints, mR.mergedEdgePoints,
                                   mL.groupIds, initialParams_)         // 2-6 ★参数只从这取
        ellL = [ellipseFit_.Execute(pts) FOR each group IN u.rectifiedPoints1]  // 2-7
        ellR = [ellipseFit_.Execute(pts) FOR each group IN u.rectifiedPoints2]
        mm = markerMatch_.Execute(ellL.centers, ellR.centers)           // 2-8
        eiL = epipolarInt_.Execute(ellL, mm.centerMatches)              // 2-9
        eiR = epipolarInt_.Execute(ellR, mm.centerMatches)
        em = edgeMatch_.Execute(eiL.ellipseResults, eiR.ellipseResults,
                                mm.centerMatches)                       // 2-10
        pr = pointRecon_.Execute(em, u.groupIds, mm.centerMatches)      // 2-11
        ☠ CHECK pr
        RETURN pr, ellL, ellR        // 椭圆中心旁路交付（3-1 输入）

    // ---------- 单帧全链便捷入口 ----------
    FUNCTION processMarkerFrame(markerGrayL, markerGrayR: Mat)
             -> {marker3D: PointReconstructCPUResult,
                 ellipseL/R: EllipseFitResults}
        fe = frontend(markerGrayL, markerGrayR)
        IF fe.frameType == LaserFrame: RETURN {skipped: true}
        RETURN compute(fe)
```

## 3. PostureCollector（B3）——流水线 1：姿态采集

```pseudo
CLASS PostureCollector:
    CONFIG:
        targetCount   = 25
        laserTubeCount = N                // 配置化（真机确认）
        confirmCycles  = 2                // 连续 2 周期同姿态才确认
        dropPolicy     = 反压（不丢帧）    // 姿态优先（调度文档 §8.2）

    PRIVATE:
        chain_          : MarkerChain
        frameFuse_      : FrameFuseCPU            // 2-12
        poseEstimate_   : PoseEstimateCPU         // 2-13
        poseTargets_    : vector<PoseTarget>      // ← CalibStore::loadPoseTargets
        globalMarkerSet_: MarkerPointSet          // 全局标记点集（首帧建立）
        collected_      : array<atomic<bool>, 25> // CAS 防重复
        collectedCount_ : atomic<int>
        pendingPoses_   : vector<CollectedPose>   // 交付产物
        lastHit_        : optional<int>           // 上一采集周期命中号（多帧确认用）
        lastCycle_      : optional<Cycle>         // 上一完整周期（回溯保存用）
        ellipseByPose_  : map<int, EllipseFitResults>  // 2-7→3-1 旁路

    // ---------- 主入口：同步运行至集齐 ----------
    FUNCTION run(progress: callback, abort: atomic<bool>&) -> Result
        loadPoseTargets / 初始化链（Warmup 期已预热的算子复用）
        WHILE collectedCount_ < targetCount AND NOT abort:
            cycle = cycleSource_.grabNext()           // ① 取周期单元（阻塞；满则反压采集线程）
                                                        //   {markerL/R, laserFrames[N], temps, ts[]}
            fe = chain_.frontend(cycle.markerL, cycle.markerR)
            IF fe.frameType == LaserFrame: CONTINUE   // 防御（不该发生）

            marker3D, ellL, ellR = chain_.compute(fe)

            // --- 2-12 配准（frame_fuse：当前帧 vs 全局标记点集）---
            IF globalMarkerSet_ 为空:
                globalMarkerSet_ = fromMarker3D(marker3D)     // 首帧建立
                ff = identityTransform()
            ELSE:
                ff = frameFuse_.Execute(toMarkerPointSet(marker3D),
                                        globalMarkerSet_)     ☠ CHECK 失败→跳过本周期

            // --- 2-13 姿态估计 + 命中 ---
            pe = poseEstimate_.Execute(ff.R, ff.T, poseTargets_)
            progress({currentPose: pe.currentPose,
                      anyMatched: pe.anyMatched,
                      collected: bitmask(collected_),
                      remaining: 清单})                        // UI 姿态引导实时反馈

            IF NOT pe.anyMatched:
                lastHit_ = none; CONTINUE

            hitId = pe.bestMatchId
            IF collected_[hitId]:                              // 已采过 → 不重复
                lastHit_ = none; CONTINUE

            // --- 多帧确认（编排层状态机，非算子）---
            IF lastHit_ == hitId
               AND 时间戳连续(lastCycle_.ts[], cycle.ts[]):     // 中间 N 管帧无丢帧
                // 确认：保存上一完整周期（回溯语义：上一周期已完整且同姿态）
                IF collected_[hitId].compare_exchange_strong(false → true):
                    pendingPoses_.push({poseId: hitId,
                                        markerL/R: lastCycle_.markerL/R,
                                        laserFrames: lastCycle_.laserFrames,
                                        temps: lastCycle_.temps,
                                        ellipseL/R: ellipseByPose_.cached})
                    ellipseByPose_[hitId] = {ellL, ellR}        // 2-7 旁路
                    collectedCount_ += 1
                    progress(集齐 {collectedCount_}/25)
            ELSE:
                // 首次命中：暂存本周期，等下一周期回溯确认
                ellipseByPose_.cache(hitId, {ellL, ellR})
                lastHit_ = hitId
                lastCycle_ = cycle
        // END WHILE
        IF abort: return aborted
        RETURN ok(pendingPoses_)

    FUNCTION takeCollected() -> vector<CollectedPose>
        RETURN move(pendingPoses_)
```

## 4. CalibrationWorkflow 重写（B4）——两阶段主控

```pseudo
CLASS CalibrationWorkflow IMPLEMENTS IWorkflow:
    PHASE: Init → Warmup → PostureCollect → Computing → Done | Error

    FUNCTION calibrationLoop():    // 现有后台线程模式保留
        // ========== Phase: Init ==========
        phase_ = Init;  notify("初始化")
        initParams = calibStore.initialParams()          ☠ CHECK 存在
        ☠ CHECK calibStore.loadPoseTargets(poseTargetsPath_)   // 25 目标
        markerChain = MarkerChain(initParams)            // 算子构造
        gpuWarmup(markerChain)                           // 显存池预分配（兼 GPU 预热）

        // ========== Phase: Warmup（阶段 0）==========
        phase_ = Warmup;  notify("预热")
        mcu = ctx_.mcu()
        mcu.enterCalib()                                 // N16 B1（假设协议就绪）
        mcu.setHeat(targetTemp)                          // 加热
        t0 = now()
        WHILE elapsed(t0) < warmupSeconds_ AND NOT abort_:
            temps = mcu.getTemperatures()                // 4 路
            tempCurve_.record(temps)                     // 温度曲线（报告数据源）
            sleep(1s); notify(倒计时)
        cameraTempBaseline_ = temps.mean()               // 温度基线

        // ========== Phase: PostureCollect（流水线 1）==========
        phase_ = PostureCollect
        collector = PostureCollector(markerChain, calibStore.poseTargets(),
                                     cycleSource=ctx_.cycleBuffer())
        r = collector.run(progress=转发UI, abort_)
        ☠ CHECK r
        poses = collector.takeCollected()                // 25×周期帧+椭圆中心

        // ========== Phase: Computing（流水线 2）==========
        phase_ = Computing

        // --- Stage3 相机标定（与激光前段并行）---
        laserFiber = async runLaserFrontEnd(poses)       // 4-1~4-4 提前线程（§5.2 前段）
        camOut = runCameraCalibChain(poses, initParams)  ☠ CHECK
        laserFiber.wait()                                 // 前段产物就位

        // --- Stage4 激光标定（串行段消费 Stage3 产物）---
        laserOut = runLaserCalibChain(laserFiber.result, camOut)  ☠ CHECK

        // ========== Phase: Done（质量门禁 + 落盘）==========
        phase_ = Done
        report = qualityGate_.evaluate(camOut, laserOut, tempCurve_,
                                       thresholds_)      // W7 判定
        writeToCalibStore(camOut, laserOut, report)
        calibStore.save("calibration.yml")
        mcu.exitCalib()                                  // N16 B0
        IF report.overall == Fault:
            state_ = Error;  result_ = {success:false, message:"质量门禁未通过", report}
        ELSE:
            state_ = Completed;  result_ = {success:true, qualityFlag: report.overall, report}
```

## 5. 流水线 2 两条链（B5/B6）

### 5.1 runCameraCalibChain（Stage3，7 算子）

```pseudo
FUNCTION runCameraCalibChain(poses: vector<CollectedPose>,
                             init: InitialParams) -> CameraChainOutput:
    // --- 3-1 逆畸变（矫正椭圆中心 → 原始畸变坐标；×25×左右）---
    invDist = InverseDistortCPU()
    FOR each pose IN poses:
        origL = invDist.Execute(pose.ellipseL.centers, init.L)   // ★与 2-6 同源参数
        origR = invDist.Execute(pose.ellipseR.centers, init.R)   // RoundTripVerify 内置自验
        origLByPose.push(origL); origRByPose.push(origR)

    // --- 3-2 内参标定（多帧原始畸变中心 + 温补后 3D 点）---
    board3D = generateBoardPoints(boardSpec)                     // 标定板 3D 点
    board3DByPose = [thermalExpand(board3D, pose.meanTemp) FOR pose]  // 按帧温度热膨胀
    intr = IntrinsicCalibCPU(params)
    intrOut = intr.Execute(origLByPose, origRByPose, board3DByPose)   ☠ CHECK rms 阈值粗查

    // --- 5-1 内参温补表 ---
    comp1 = IntrinsicCompensateCPU().Execute(intrOut.toCameraIntrinsics(refTemp=baseline))

    // --- 3-3 外参标定（←3-2 内参）---
    extr = ExtrinsicCalibCpu(params{rotateRight180=true})
    extOut = extr.Execute(origLByPose, origRByPose, board3D, intrOut)  ☠ CHECK

    // --- 5-2 外参温补表 ---
    comp2 = ExtrinsicCompensateCPU().Execute(extOut.toCameraExtrinsics())

    // --- 3-4 立体矫正 ---
    rect = StereoRectifyCpu(params)
    rectOut = rect.Execute(intrOut, extOut)                     ☠ CHECK

    // --- 3-5 温度表 1 ★ ---
    tt = StereoRectifyTempTableCpu(params{refTemp=baseline, cte=23.6e-6, step=0.2})
    tempTable1 = tt.Execute(intrOut, extOut)                    ☠ CHECK

    RETURN {intrOut, extOut, rectOut, tempTable1, comp1, comp2}
```

### 5.2 runLaserCalibChain（Stage4，PJC 路线）

```pseudo
// ---- 前段（可与 Stage3 并行的独立线程）----
FUNCTION runLaserFrontEnd(poses) -> LaserFrontOutput:
    FOR each pose IN poses:                        // 25 姿态 × 各自激光管帧
        FOR each laserFrame IN pose.laserFrames:
            me = MaskExtractCUDA().Execute(laserFrame.L)        // 4-1
            ccl = RegionAnalyzerCUDA().Execute(me.d_cleanedMask) // 4-2
            lab = LaserLabelCUDA().Execute(ccl.d_labeledMask)    // 4-3 按Y编号
            st  = StegerExtractorCUDA(ByLabel).Execute(me.d_grayImage,
                                                      lab.d_labeledMask) // 4-4
            frontOutput[pose][tube].push({st.d_centerPoints, st.d_line_ids})
    RETURN frontOutput                                        // GpuMat 池暂存

// ---- 串行段（消费 Stage3 产物）----
FUNCTION runLaserCalibChain(front: LaserFrontOutput, cam: CameraChainOutput) -> LaserChainOutput:
    f_main, cx, cy = decomposeK(cam.intrOut.stereoK)            // f/主点 ← stereoK 派生

    FOR each pose IN poses:
        FOR each tube IN pose:
            // --- 4-5 去畸变（★标定后内参，区别于 2-6 的初始参数）---
            ud = UndistortPointsCuda().Execute(front[pose][tube].d_centerPoints,
                                               cam.intrOut.calibrated)
            // --- 4-6 极线插值（lineIdCheck=true 标定模式）---
            ep = EpipolarInterpCuda(lineIdCheck=true).Execute(ud.d_rectifiedPoints, ud.d_line_ids)
            interpL[pose][tube] = ep                            // 左
            （右路同 4-1~4-6 后）interpR[pose][tube] = ...
        // --- 4-7 激光线匹配（标定版 ByLabel）---
        lm = LaserMatchCuda().Execute(interpL[pose], interpR[pose])
        // --- 4-8 三维重建（Q ← 3-4）---
        lr = LaserReconstructCuda().Execute(lm.d_matchedL, lm.d_matchedR,
                                            lm.d_matched_line_ids, cam.rectOut.Q)
        poses3d[pose] = lr.d_points3d

    // --- [编排层聚合]（非算子）---
    posesCloud = [download(poses3d[p]).groupBy(p) FOR p]        // D2H ↓，vector<PosePointSet>

    // --- PJC 联合优化（D2=a 定案链路）---
    pjc = ProjectorJointCalib(params{f, principalPoint=(cx,cy),
                                     initialT=机械公差(80,3,3)mm})
    pjcOut = pjc.Execute({posesCloud})                          ☠ CHECK 条件数/rms 诊断

    // --- 4-12 平面映射表 ---
    pm = PlaneMap({virtualK=[f,0,cx;0,f,cy;0,0,1], virtualR=I,  // R=I 是修正（非假设）
                   virtualT=pjcOut.projectorT})
    pmOut = pm.Execute(cam.stereoCalibration)

    // --- 4-13 温度表 2 ★ ---
    tt = PlaneMapTempTable({virtualK/R/T ← PJC, cam ← 3-4, cte})
    tempTable2 = tt.Execute()                                   ☠ CHECK

    // --- 5-3 激光外参温补表 ---
    le = LaserExtrinsicCompensateCPU().Execute(
            {R=I, T=pjcOut.projectorT},                          // 虚拟→左
            {derive(cam.extOut)})                                // 虚拟→右（编排推导）

    RETURN {pjcOut, pmOut, tempTable2, le}
```

## 6. CalibQualityGate（B7）

```pseudo
FUNCTION evaluate(cam, laser, tempCurve, th: QualityThresholds) -> CalibQualityReport:
    report = {}
    report.intrinsicRmsL = cam.intrOut.left.rms_error
    report.intrinsicRmsR = cam.intrOut.right.rms_error
    report.stereoRms     = cam.extOut.stereoReprojError
    report.validPoseCount = 收集期确认姿态数（应=25）
    report.tempTableComplete = [tempTable1 无阶梯缺口, tempTable2 无缺口]
    report.pjcDiagnostics = laser.pjcOut.{条件数, rms, improvementRatio}
    report.targetTableVersion = 姿态表版本串
    report.tempCurveSummary = tempCurve 统计（M5 报告）

    // 判定（阈值外置，W7-1 实测后填）
    faults = []
    IF report.intrinsicRmsL > th.intrinsicRmsMax: faults.push("左内参RMS超限")
    IF report.intrinsicRmsR > th.intrinsicRmsMax: faults.push("右内参RMS超限")
    IF report.stereoRms > th.stereoRmsMax: faults.push("立体RMS超限")
    IF report.validPoseCount < th.minValidPoses: faults.push("有效姿态不足")
    IF NOT report.tempTableComplete[0] OR NOT [1]: faults.push("温度表缺口")
    IF laser.pjcOut.conditionNumber > th.pjcCondMax: faults.push("PJC病态")

    report.overall = faults.empty ? Normal
                                   : (faults 全为软指标 ? Degraded : Fault)
    RETURN report
```

```pseudo
FUNCTION writeToCalibStore(cam, laser, report):
    s = ctx_.calibStore()
    // 相机链产物
    s.setCameraMatrixL/R(cam.intr); s.setDistCoeffsL/R(...)
    s.setExtrinsicR/T(cam.ext);     s.setR1/R2/P1/P2/Q(cam.rect)
    s.setStereoRectifyTempTable(cam.tempTable1)                 // ★W2 闭环
    // 激光链产物
    s.setProjectorT(laser.pjcOut.projectorT)
    s.setEmissionCurve(laser.pjcOut.emissionCurve)
    s.setLaserPlaneMapTempTable(laser.tempTable2)               // ★W2 闭环
    // 质量
    s.setOverallQuality(report.overall)
    // 全部落盘由调用方 s.save() 完成
```

## 8. 关键横切点备忘

| 点 | 规则 |
|---|---|
| 2-6 ⇄ 3-1 | **只从 `CalibStore::initialParams()` 取**——两处调用的参数一致性由单一来源保证，不各自传参 |
| 4-5 参数 | 用 **3-2 标定后**内参（与 2-6 初始参数刻意不同，v2.2 4-5 标注） |
| abort | 每个循环体/每姿态边界检查 `abort_`，Warmup 每秒、Collector 每周期、Chain 每姿态 |
| GpuMat 生命周期 | 前段产物暂存显存池（25 姿态×N 管预算），串行段消费后释放；Warmup 期预分配 |
| 时间戳连续判据 | `cycle.ts[]` 单调且相邻差 ≤ 帧周期×1.5（多帧确认第 3 步） |
| 反压语义 | cycleSource 满时阻塞采集线程（不丢帧）；Collector 端 grabNext 无超时丢帧路径 |
