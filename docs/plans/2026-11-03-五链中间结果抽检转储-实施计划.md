**Step 1:** 写 `fc_probe.h`：
```cpp
#pragma once
// ============================================================================
// fc_probe.h — 五链中间结果抽检转储（design: 2026-11-03）
// 帧号命中 probe.frames 时，把该帧 CycleProducts 全链中间结果按 scan_replay
// 沙箱命名（README §5）落盘 diagnostics/fc_probe/cycle_NNNN/。默认关＝零开销。
// ============================================================================
#include <cstdint>
#include <set>
#include <string>
#include "harness.h"   // scr::CycleProductsCompat

namespace jmw::probe {

struct ProbeConfig {
    bool enabled = false;
    long maxFrames = 1000;                 // app 层停采用；07 只读 frames
    std::set<long> frames;                 // 1 起的流水线帧号
    static ProbeConfig load();             // 读 config/fc_dump.json probe 段
};

class FrameProbe {
public:
    void configure(const ProbeConfig& c) { cfg_ = c; }
    bool enabled() const { return cfg_.enabled; }
    /// idx＝fc slot 序号（0 起）→ 帧号 idx+1；命中则落全套产物
    void maybeDump(long idx, const scr::CycleProductsCompat& p);
    /// 全程逐帧位姿追加（probe 开启时每帧一行，含非命中帧）
    void appendRt(long idx, const scr::CycleProductsCompat& p);
private:
    ProbeConfig cfg_;
};

} // namespace jmw::probe
```

**Step 2:** 写 `fc_probe.cpp`。**先核对字段**：打开以下头文件确认结果结构字段名后落码（沙箱 harness.cpp 同款取法的现成参照）：

- `modules/09_OperatorLib/core/vision/region_analyze_cuda.h`（CCL 组件列表字段）
- `modules/09_OperatorLib/scanning/laser/laser_match_scan_v3_split.h`（匹配对字段）
- `modules/09_OperatorLib/scanning/laser/laser_reconstruct_cpu.h`（3D 点字段）
- `modules/09_OperatorLib/core/marker/`（split/zernike/merge/undistort/ellipse/marker_match/conic 各结果字段）
- 已核实：`StegerResult.centerPoints`（map<int,vector<Point2f>>）＋`totalPointCount`（steger_types.h:118）；`EpipolarInterpResult.d_interpPoints/d_interp_line_ids/interpCount`（epipolar_types.h:25，GpuMat 需下载→用 artifacts `saveGpuMatP2`）

骨架（按沙箱 harness.cpp 落盘段逐条对应，`art.h/art.`＝搬来的 artifacts）：
```cpp
#include "fc_probe.h"
#include "artifacts.h"
#include <filesystem>
#include <fstream>
#include <nlohmann/json.hpp>
#include "jmw_logging.h"

namespace jmw::probe {
namespace fs = std::filesystem;
namespace art = scr;   // artifacts 工具箱 namespace scr

ProbeConfig ProbeConfig::load() {
    ProbeConfig c;
    try {
        std::ifstream f(fs::path("config") / "fc_dump.json");
        if (!f) return c;
        nlohmann::json j; f >> j;
        if (j.contains("probe")) {
            const auto& p = j["probe"];
            c.enabled = p.value("enabled", false);
            c.maxFrames = p.value("maxFrames", 1000);
            for (const auto& e : p.value("frames", std::vector<long>{}))
                c.frames.insert(e);
        }
    } catch (...) { /* 开关件失败静默 */ }
    return c;
}

void FrameProbe::maybeDump(long idx, const scr::CycleProductsCompat& p) {
    if (!cfg_.enabled || cfg_.frames.empty()) return;
    const long frameNo = idx + 1;
    if (cfg_.frames.count(frameNo) == 0) return;
    try {
        char nb[16]; std::snprintf(nb, sizeof(nb), "cycle_%04ld", frameNo);
        const fs::path dir = fs::path("diagnostics") / "fc_probe" / nb;
        fs::create_directories(dir);
        const std::string d = dir.string();
        // 0 原图（算法几何 1688x1400 口径；pre-crop 原图由既有 fc_frames 机制另存）
        art::savePng(d + "/0-frames.png", /* L|隔条|R 并合，仿沙箱 hconcat */ {});
        // L0/M0 掩膜（叠加着色；GpuMat 则先下载）
        art::saveMaskOverlayPng(d + "/L0-laser_mask.png", p.frameGrayL, /* sepL 激光掩膜 */ {}, {0,255,0});
        art::saveMaskOverlayPng(d + "/M0-marking_mask.png", p.frameGrayL, /* sepL 标志点掩膜 */ {}, {0,180,255});
        // M1 CCL（伪彩＋编号，仿沙箱；数据源 p.cclL/p.cclR）
        // L1 steger：centerPoints 摊平 CSV＋叠加图；L2 undistCudaL/R；
        // L3 interpL/R（GpuMat→saveGpuMatP2）；L4 matchV3（同线同色）；
        // L5 recon → savePlyXYZf
        // M2 splitL/R、M3 zernL/R、M4 mergeL/R、M5 undistCpu、M6 ellL/ellR(+json)、
        // M7 mm、M10 conic(savePlyXYZf)
        // M11 registration.json：R/T/regSource/regFlow 诊断
        // summary.json：markersDetected/Valid、laserMatched/Points3D、rowResidMedPx、stepSuccess_
        JMW_LOG_WARN("07-FiveChain", "[probe] 帧 {} 中间产物已落盘 → {}", frameNo, d);
    } catch (const std::exception& e) {
        JMW_LOG_WARN("07-FiveChain", "[probe] 帧 {} 落盘失败: {}", frameNo, e.what());
    }
}

void FrameProbe::appendRt(long idx, const scr::CycleProductsCompat& p) {
    if (!cfg_.enabled) return;
    try {
        const fs::path dir = fs::path("diagnostics") / "fc_probe";
        fs::create_directories(dir);
        std::ofstream rt(dir / "rt.csv", std::ios::app);
        rt << (idx + 1) << ',' << p.regSource << ','
           << p.R[0] << ',' << p.R[1] << ',' << p.R[2] << ','
           << p.R[3] << ',' << p.R[4] << ',' << p.R[5] << ','
           << p.R[6] << ',' << p.R[7] << ',' << p.R[8] << ','
           << p.T[0] << ',' << p.T[1] << ',' << p.T[2] << '\n';
    } catch (...) { }
}
} // namespace jmw::probe
```
（骨架中 `/* */` 处按 Step 2 核对的字段实名补齐；每类产物写法逐条对照沙箱 `harness.cpp` 中同名产物落盘段——搜 `L0-laser_mask`、`M1-ccl` 等文件名即达。）
