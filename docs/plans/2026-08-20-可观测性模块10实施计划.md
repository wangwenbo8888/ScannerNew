// §4.1 转换矩阵（S6 无故障边=免疫；S6/S7 无断连边=例外；SelfCheckPassed 特判不在表内）

    {SystemState::Init,            EventType::FaultOccurred,      SystemState::FaultSelfCheck},
    {SystemState::Standby,         EventType::CalibStarted,       SystemState::Calibrating},

    {SystemState::Standby,         EventType::ScanStarted,        SystemState::ScanMarkerLaser},   // param=1
    {SystemState::Standby,         EventType::PostProcessStarted, SystemState::PostProcessing},
    {SystemState::Standby,         EventType::FaultOccurred,      SystemState::FaultSelfCheck},
    {SystemState::Standby,         EventType::DeviceDisconnected, SystemState::Init},
    {SystemState::Calibrating,     EventType::CalibFinished,      SystemState::Standby},
    {SystemState::Calibrating,     EventType::FaultOccurred,      SystemState::FaultSelfCheck},
    {SystemState::Calibrating,     EventType::DeviceDisconnected, SystemState::Init},
    {SystemState::ScanMarker,      EventType::ScanStopped,        SystemState::Standby},
    {SystemState::ScanMarker,      EventType::FaultOccurred,      SystemState::FaultSelfCheck},
    {SystemState::ScanMarker,      EventType::DeviceDisconnected, SystemState::Init},
    {SystemState::ScanMarkerLaser, EventType::ScanStopped,        SystemState::Standby},
    {SystemState::ScanMarkerLaser, EventType::FaultOccurred,      SystemState::FaultSelfCheck},
    {SystemState::ScanMarkerLaser, EventType::DeviceDisconnected, SystemState::Init},
    {SystemState::PostProcessing,  EventType::PostProcessFinished,SystemState::Standby},
    {SystemState::PostProcessing,  EventType::DeviceDisconnected, SystemState::Init},  // §4.5 免疫后此行删除——见 Step5 校验
    {SystemState::FaultSelfCheck,  EventType::SelfCheckPassed,    SystemState::Standby}, // 占位，实际走 lastHealthy_ 特判

}

Result StateMachine::transition(EventType event, int64_t param) {
    SystemState from = state_.load(std::memory_order_acquire);
    for (;;) {
        SystemState to{};
        if (!resolveNext(from, event, param, to)) {
            spdlog::warn("[StateMachine] 非法转换: {} + {}", stateToString(from), (int)event);
            return Result::fail("非法状态转换");
        }
        if (state_.compare_exchange_weak(from, to, std::memory_order_acq_rel, std::memory_order_acquire)) {
            if (to == SystemState::FaultSelfCheck && from != SystemState::FaultSelfCheck)
                lastHealthy_.store(from);                      // S7 进入记账
            notifyChange(from, to);                            // 锁外
            if (eventBus_) { /* publish StateChanged(param1=from,param2=to) */ }
            return Result::ok();
        }   // CAS 失败：from 已更新为最新值，循环重判
    }
}

bool StateMachine::resolveNext(SystemState from, EventType ev, int64_t param, SystemState& to) {
    if (ev == EventType::SelfCheckPassed && from == SystemState::FaultSelfCheck) {
        to = lastHealthy_.load();
        if (to == SystemState::FaultSelfCheck || to == SystemState::PostProcessing) to = SystemState::Standby; // 兜底/永不回 S6
        return true;
    }
    for (const auto& e : kEdges) {
        if (e.from != from || e.ev != ev) continue;
        if (ev == EventType::ScanStarted && e.to == SystemState::ScanMarker && param != 0) continue;
        if (ev == EventType::ScanStarted && e.to == SystemState::ScanMarkerLaser && param != 1) continue;
        to = e.to; return true;
    }
    return false;
}

```cpp
// CommandGate.h — 统一命令通道（submit 触发 / notifyCompleted 收尾切态；设计方案 §3）
#include "StateMachine.h"
#include <functional>
#include <mutex>
#include <string>
#include <unordered_map>

namespace Scanner::service {

class CommandGate {
public:
    using Handler = std::function<Result()>;                    // 仅点火，毫秒级返回
    using Precondition = std::function<Result()>;               // 业务前置谓词（如参数就绪）
    struct Spec {
        std::string name;
        std::string gateOp;            // canOperate 操作名（""=不查门禁，内部命令）
        EventType startedEvent{};      // 0=触发类不切态（finish_scan 型）
        SystemState startedTo{};       // 切态目标（startedEvent==0 时忽略）
        EventType finishedEvent{};     // notifyCompleted 切态用
        Precondition pre;
        Handler handler;
    };

    explicit CommandGate(StateMachine* sm) : sm_(sm) {}

    Result registerCommand(Spec spec);
    Result submit(const std::string& name, int64_t payload = 0);   // 三关→切态→点火→失败回滚
    Result notifyCompleted(const std::string& name, bool ok);      // 唯一切态收尾口

private:
    StateMachine* sm_;
    std::mutex mtx_;
    std::unordered_map<std::string, Spec> cmds_;
};
} // namespace Scanner::service
```

submit 实现要点（全在 mutex 内完成三关+切态，handler 点火在锁外、失败回滚重新入锁）：
① 查注册 → ② `gateOp` 空 || `sm_->canOperate(gateOp)` → ③ `pre` 空 || pre().ok → ④ startedEvent≠0 则 `sm_->transition(startedEvent, payload)`（失败=有并发抢先，返回 fail）→ ⑤ handler() → ⑥ 同步失败且此前切过态：`sm_->transition(finishedEvent)` 回滚 + publish CommandRejected + ERROR 日志 → ⑦ startedEvent≠0 时 publish startedEvent。
notifyCompleted：查注册 → `sm_->transition(finishedEvent)`（ok 参数 false 时也切回 S2 但记 ERROR）。
