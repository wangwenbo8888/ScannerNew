```cpp
// —— 场景（全局系真值）——
struct SimScene {
    std::vector<calib::MarkerCloudPoint> markers;  // ← markers_30.ply（N=30）
    std::vector<cv::Point3f> laser;                // ← 导入点云 stash（M 可达百万级）
    static SimScene builtin(uint32_t seed = 42);   // fallback：30 标志点＋6000 点圆柱面片
    static Scanner::Result loadMarkersFromPly(const std::string& path);  // fileio::importPLY
    static Scanner::Result save(const std::string& path) const;          // JSON（调试留档可选拷贝）
    static Scanner::Result load(const std::string& path, SimScene& out);
};

// —— 轨迹与抖动 ——
struct SimTrajParams {
    double rotDegPerFrame = 0.3;     // 整体移动：绕 axis 每帧转角
    double transMmPerFrame = 0.5;    // 整体移动：每帧平移（沿 x）
    cv::Vec3d rotAxis{0, 1, 0};
    double jitterSigmaMm = 0.05;     // 激光逐点抖动 σ（< marker voxel 0.5mm 的 1/5）
    size_t laserPerFrame = 100000;   // 每帧激光子采样（大点云不可能逐帧全量上传；
                                     //   逐帧随机子集≈逐帧部分表面观测，兼真实感与带宽）
    uint64_t maxFrames = 600;        // 默认 600 帧＝20s@30fps
};

// —— 逐帧源（单线程独占调用）——
class SimScanSource {
public:
    SimScanSource(SimScene s, SimTrajParams p = {});
    void reset();
    /// 产下一帧 FrameResult：G_k = T(k·t)·R(k·θ)；
    /// markers_k = G_k⁻¹·M（无逐点噪声，30 点全量）；
    /// laser_k = G_k⁻¹·(P 的随机子集) + N(0,σ²)（每帧子集重抽）；
    /// R/T = G_k（设备系→全局系，与 FuseConsumer fuse(m,R,T) 约定一致）；
    /// laser host→GpuMat(CV_32FC3 1×N) 上传为 GpuPointCloudBlock（池外直构）。
    bool next(FrameResult& out);     // false＝maxFrames 到
};
```

实现要点：
- 子采样＝`mt19937` 均匀分布不重复抽 `laserPerFrame` 个下标（M 小于该值则全量）；抖动 `normal_distribution`，seed 独立于场景（可复现）；
- R/T 方向存疑时**先只实现到可被 T3 用例 0 证伪**（见风险 R1）；
- CUDA 守卫：`#ifdef JMW_BUILD_CUDA` 包激光上传段；无 CUDA 构建仅标志点路径（与 07 现状一致）；
- 日志 tag `07-SimScan`：构造（场景规模/子采样量）/每 100 帧心跳。
